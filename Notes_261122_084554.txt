  getState() { return { conflicts: Cloud.conflicts.slice(), error: Cloud.error, saving: Object.keys(Cloud.pending).length > 0 }; },
  notify() { const s = Cloud.getState(); Cloud.listeners.forEach(f => { try { f(s); } catch (e) {} }); },
  reset() {
    Object.values(Cloud.timers).forEach(clearTimeout);
    Cloud.uid = null; Cloud.cache = {}; Cloud.rev = {}; Cloud.pending = {}; Cloud.timers = {};
    Cloud.tries = {}; Cloud.conflicts = []; Cloud.error = ""; Cloud.notify();
  },
  ref(k) { return fbDb.collection("users").doc(Cloud.uid).collection("data").doc(k); },
  ts() { return firebase.firestore.FieldValue.serverTimestamp(); },

  queue(k, v) {
    Cloud.pending[k] = v;
    clearTimeout(Cloud.timers[k]);
    Cloud.timers[k] = setTimeout(() => Cloud.save(k), Cloud.debounce);
    Cloud.notify();
  },

  // Ghi một khóa bằng transaction có kiểm tra phiên bản (rev) để không ghi đè dữ liệu mới hơn từ thiết bị khác
  async save(k) {
    if (!(k in Cloud.pending) || !Cloud.uid) return true;
    const v = Cloud.pending[k];
    if (v.length > MAX_VALUE_LEN) {
      Cloud.error = "Dữ liệu \"" + k + "\" quá lớn để lưu lên máy chủ."; Cloud.notify(); return false;
    }
    const ref = Cloud.ref(k);
    try {
      const rev = await fbDb.runTransaction(async (t) => {
        const s = await t.get(ref);
        const cur = s.exists ? (s.data().rev || 0) : 0;
        if (cur !== (Cloud.rev[k] || 0)) { const e = new Error("conflict"); e.conflict = true; throw e; }
        t.set(ref, { value: v, rev: cur + 1, updatedAt: Cloud.ts() });
        return cur + 1;
      });
      Cloud.rev[k] = rev; Cloud.tries[k] = 0;
      if (Cloud.pending[k] === v) delete Cloud.pending[k];
      Cloud.conflicts = Cloud.conflicts.filter(x => x !== k);
      Cloud.error = ""; Cloud.notify();
      return true;
    } catch (e) {
      if (e && e.conflict) {
        if (!Cloud.conflicts.includes(k)) Cloud.conflicts.push(k);
      } else if (e && e.code === "permission-denied") {
        Cloud.error = "Không có quyền ghi dữ liệu. Hãy kiểm tra Firestore Security Rules và đăng nhập lại.";
      } else {
        Cloud.error = "Không lưu được lên máy chủ (mạng?). Hệ thống sẽ tự thử lại.";
        Cloud.tries[k] = (Cloud.tries[k] || 0) + 1;
        clearTimeout(Cloud.timers[k]);
        Cloud.timers[k] = setTimeout(() => Cloud.save(k), Math.min(30000, Cloud.retryBase * Math.pow(2, Cloud.tries[k] - 1)));
      }
      Cloud.notify();
      return false;
    }
  },

  async flush() {
    const keys = Object.keys(Cloud.pending);
    keys.forEach(k => clearTimeout(Cloud.timers[k]));
    await Promise.all(keys.map(k => Cloud.save(k)));
  },

    return { up, skip };
  },

  }
